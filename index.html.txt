function carregarDados(){


  try{


    memories=JSON.parse(
      localStorage.getItem(STORAGE_MEMORIES)||"[]"
    );


    users=JSON.parse(
      localStorage.getItem(STORAGE_USERS)||"[]"
    );


    const email=localStorage.getItem(STORAGE_CURRENT);


    if(email){


      currentUser=users.find(
        user=>
          user.email.toLowerCase()===
          email.toLowerCase()
      )||null;


    }


  }catch(error){


    memories=[];
    users=[];
    currentUser=null;


  }
}


function salvarDados(){


  localStorage.setItem(
    STORAGE_MEMORIES,
    JSON.stringify(memories)
  );


  localStorage.setItem(
    STORAGE_USERS,
    JSON.stringify(users)
  );


  if(currentUser){


    localStorage.setItem(
      STORAGE_CURRENT,
      currentUser.email
    );


  }
}

function login(event){


  event.preventDefault();


  const email=
    document.getElementById("loginEmail")
    .value.trim().toLowerCase();


  const password=
    document.getElementById("loginPassword")
    .value;


  const user=users.find(
    item=>
      item.email.toLowerCase()===email &&
      item.password===password
  );


  if(!user){


    alert("E-mail ou senha incorretos.");
    return;


  }


  user.lastAccessAt=Date.now();


  if(!user.followers)
    user.followers=[];


  if(!user.following)
    user.following=[];


  currentUser=user;


  salvarDados();


  currentPage="home";


  render();
}


function signup(event){


  event.preventDefault();


  const name=
    document.getElementById("signupName")
    .value.trim();


  const username=
    document.getElementById("signupUsername")
    .value.trim()
    .replace(/^@/,"");


  const email=
    document.getElementById("signupEmail")
    .value.trim()
    .toLowerCase();


  const password=
    document.getElementById("signupPassword")
    .value;


  const photoInput=
    document.getElementById("signupPhoto");


  if(
    users.some(
      user=>user.email.toLowerCase()===email
    )
  ){


    alert("Este e-mail já está cadastrado.");
    return;


  }


  if(
    users.some(
      user=>
        user.username.toLowerCase()===
        username.toLowerCase()
    )
  ){


    alert("Este nome de usuário já está em uso.");
    return;


  }


  function criarUsuario(photo){


    const user={


      id:Date.now().toString(),
      name,
      username,
      email,
      password,
      photo:photo||"",
      followers:[],
      following:[],
      createdAt:Date.now(),
      lastAccessAt:Date.now()


    };


    users.push(user);


    currentUser=user;


    salvarDados();


    currentPage="home";


    render();


    alert("Conta criada com sucesso! 🌊");


  }


  if(photoInput.files[0]){


    const reader=new FileReader();


    reader.onload=e=>{
      criarUsuario(e.target.result);
    };


    reader.readAsDataURL(
      photoInput.files[0]
    );


  }else{


    criarUsuario("");


  }
}

function memoryCard(memory){


  if(!memory.likedBy)
    memory.likedBy=[];


  if(!memory.dislikedBy)
    memory.dislikedBy=[];


  if(!memory.comments)
    memory.comments=[];


  const liked=
    memory.likedBy.includes(
      currentUser.email
    );


  const disliked=
    memory.dislikedBy.includes(
      currentUser.email
    );


  const isOwner=
    memory.userEmail.toLowerCase()===
    currentUser.email.toLowerCase();


  const author=
    users.find(
      user=>
        user.email.toLowerCase()===
        memory.userEmail.toLowerCase()
    );


  const isFollowing=
    currentUser.following &&
    currentUser.following.includes(
      memory.userEmail
    );


  const photo=
    author&&author.photo
    ?`
      <div class="avatar">
        <img
          src="${author.photo}"
          alt="Foto de perfil"
        >
      </div>
    `
    :`
      <div class="avatar">
        ${initials(memory.user)}
      </div>
    `;


  return`
  <article class="memory-card">


    <img
      class="memory-image"
      src="${memory.photo}"
      alt="${escapeHTML(memory.title)}"
    >


    <div class="memory-content">


      <div class="memory-user">


        ${photo}


        <div class="user-info">


          <strong>
            ${escapeHTML(memory.user)}
          </strong>


          <small>
            @${escapeHTML(memory.username||"")}
          </small>


        </div>


        ${
          !isOwner
          ?`
            <button
              class="action-btn ${
                isFollowing?"active":""
              }"
              onclick="toggleFollow('${escapeJS(memory.userEmail)}')"
              style="margin-left:auto">


              ${
                isFollowing
                ?"✓ Seguindo"
                :"+ Seguir"
              }


            </button>
          `
          :""
        }


      </div>


      <h3 class="memory-title">
        ${escapeHTML(memory.title)}
      </h3>


      <div class="memory-meta">
        ${cityEmoji(memory.city)}
        ${escapeHTML(memory.city)}
        •
        ${formatDate(memory.date)}
      </div>


      <div class="emotion">
        ${escapeHTML(memory.emotion)}
      </div>


      <p class="memory-text">
        ${escapeHTML(memory.text)}
      </p>


      <div class="memory-actions">


        <button
          class="action-btn ${liked?"active":""}"
          onclick="toggleLike('${memory.id}')">
          👍 ${memory.likedBy.length}
        </button>


        <button
          class="action-btn ${disliked?"active":""}"
          onclick="toggleDislike('${memory.id}')">
          👎 ${memory.dislikedBy.length}
        </button>


        <button
          class="action-btn"
          onclick="adicionarComentario('${memory.id}')">
          💬 ${memory.comments.length}
        </button>


        <button
          class="action-btn"
          onclick="openCity('${escapeJS(memory.city)}')">
          🗺️ Mapa
        </button>


        ${
          isOwner
          ?`
            <button
              class="action-btn"
              onclick="deletarMemoria('${memory.id}')">
              🗑️ Excluir
            </button>
          `
          :`
            <button
              class="action-btn"
              onclick="denunciarMemoria('${memory.id}')">
              ⚠️ Denunciar
            </button>
          `
        }


      </div>


      ${
        memory.comments.length

            </strong>

                  </strong>


                  <br>


                  ${escapeHTML(comment.text)}


                </div>
              `).join("")
            }


          </div>
        `
        :""

}

  if(!memory)return;


  if(

  ){


    alert(
      "Você não pode avaliar sua própria publicação."
    );


    return;
  }


  if(!memory.likedBy)
    memory.likedBy=[];


  if(!memory.dislikedBy)
    memory.dislikedBy=[];


  const index=
    memory.likedBy.indexOf(
      currentUser.email
    );


  if(index>=0){


    memory.likedBy.splice(index,1);


  }else{


    memory.dislikedBy=
      memory.dislikedBy.filter(
        email=>email!==currentUser.email
      );


    memory.likedBy.push(
      currentUser.email
    );


  }


  salvarDados();
  renderPage();
}

/* COMENTÁRIOS */


function adicionarComentario(id){


  const memory=
    memories.find(item=>item.id===id);


  if(!memory)return;


  const text=
    prompt("Escreva seu comentário:");


  if(!text||!text.trim())

  if(!memory.comments)
    memory.comments=[];


  memory.comments.push({


    id:Date.now().toString(),


    user:currentUser.name,


    userEmail:currentUser.email,


    text:text.trim(),


    createdAt:Date.now()


  });


  salvarDados();
  renderPage();
}


/* SEGUIR */


function toggleFollow(email){


  if(
    email.toLowerCase()===
    currentUser.email.toLowerCase()
  ){
    return;
  }


  const target=
    users.find(
      user=>
        user.email.toLowerCase()===
        email.toLowerCase()
    );


  if(!target)return;


  if(!currentUser.following)
    currentUser.following=[];


  if(!target.followers)
    target.followers=[];


  const index=
    currentUser.following.indexOf(
      target.email
    );


  if(index>=0){


    currentUser.following.splice(index,1);


    target.followers=
      target.followers.filter(
        follower=>
          follower!==currentUser.email
      );


  }else{


    currentUser.following.push(
      target.email
    );


    target.followers.push(
      currentUser.email
    );


  }


  salvarDados();
  renderPage();
}

function cityEmoji(city){

function initials(name){


  if(!name)return"🌊";


  return name
    .split(" ")
    .slice(0,2)
    .map(part=>part.charAt(0))
    .join("")
    .toUpperCase();
}


function formatDate(date){


  if(!date)return"";


  const parts=date.split("-");


  if(parts.length!==3)
    return escapeHTML(date);


  return`
    ${parts[2]}/${parts[1]}/${parts[0]}
  `;
}


function formatDateTime(timestamp){


  if(!timestamp)return"—";


  return new Date(timestamp)
    .toLocaleString(
      "pt-BR",
      {
        dateStyle:"short",
        timeStyle:"short"
      }
    );
}


function escapeHTML(value){


  return String(value??"")
    .replace(/&/g,"&amp;")
    .replace(/</g,"&lt;")
    .replace(/>/g,"&gt;")
    .replace(/"/g,"&quot;")
    .replace(/'/g,"&#039;");
}


function escapeJS(value){


  return String(value??"")
    .replace(/\\/g,"\\\\")
    .replace(/'/g,"\\'")
    .replace(/"/g,'\\"')
    .replace(/\n/g,"\\n")
    .replace(/\r/g,"");


}
